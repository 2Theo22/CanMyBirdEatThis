CREATE TABLE Food (
    FoodName VARCHAR(50) UNIQUE NOT NULL,
    Danger VARCHAR(20) NOT NULL
        CHECK (Danger IN ('can eat', 'should not eat', 'dangerous', 'NA')),
    FoodID INTEGER PRIMARY KEY UNIQUE NOT NULL,
    SmallPortionWarning VARCHAR(20)
        CHECK (SmallPortionWarning IN ('Only small portions!'),
    Safety level VARCHAR(10) NOT NULL
        CHECK (Safety in ('Safe', 'Ok', 'Unsafe', 'NA')),
    FoodType VARCHAR (20)
        CHECK (FoodType in ('Vegetable', 'Fruits', 'Flowers/Plants', 'Meat', 'Nuts'),
    CertainDeadWarning VARCHAR(20)
        CHECK (CertainDeadWarning in ('Certain Dead!')
);
